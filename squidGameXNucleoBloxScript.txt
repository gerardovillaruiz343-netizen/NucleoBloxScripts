local enabled = false
local size = 12
local hitboxTransparency = 0.9
local hitboxColor = Color3.fromRGB(255, 0, 0)

local originalRoots = setmetatable({}, { __mode = "k" })

local function getRoot(player)
    local character = player.Character
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function restoreRoot(root)
    local saved = originalRoots[root]
    if not saved then
        return
    end
    originalRoots[root] = nil
    if root.Parent then
        root.Size = saved.Size
        root.Transparency = saved.Transparency
        root.Color = saved.Color
        root.Material = saved.Material
        root.CanCollide = saved.CanCollide
    end
end

local function applyRoot(root)
    if not originalRoots[root] then
        originalRoots[root] = {
            Size = root.Size,
            Transparency = root.Transparency,
            Color = root.Color,
            Material = root.Material,
            CanCollide = root.CanCollide,
        }
    end
    local target = Vector3.new(size, size, size)
    if root.Size ~= target then
        root.Size = target
    end
    root.Transparency = hitboxTransparency
    root.Color = hitboxColor
    root.Material = Enum.Material.SmoothPlastic
    root.CanCollide = false
end

local function refresh()
    for _, player in Players:GetPlayers() do
        if player ~= LocalPlayer then
            local root = getRoot(player)
            if root then
                if enabled then
                    applyRoot(root)
                else
                    restoreRoot(root)
                end
            end
        end
    end
end

RunService.Heartbeat:Connect(function()
    if enabled then
        refresh()
    end
end)

    for _, state in blockedStates do

local function startAntiRagdoll()
    if antiConnection then
        return
    end
    antiConnection = RunService.Heartbeat:Connect(function()
        local character = LocalPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not humanoid then
            return
        end
        setStates(humanoid, false)
        if humanoid.PlatformStand and not LocalPlayer:GetAttribute("FlyActive") then
            humanoid.PlatformStand = false
        end
        local state = humanoid:GetState()
        if state == Enum.HumanoidStateType.Ragdoll or state == Enum.HumanoidStateType.FallingDown then
            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        for _, descendant in character:GetDescendants() do
            if descendant:IsA("Motor6D") then
                descendant.Enabled = true
            elseif descendant:IsA("BallSocketConstraint") then
                descendant.Enabled = false
            end
        end
    end)

    end
    local character = LocalPlayer.Character

local function startFly()
    stopFly()
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then
        return
    end
    LocalPlayer:SetAttribute("FlyActive", true)
    humanoid.PlatformStand = true
    flyVelocity = Instance.new("BodyVelocity")
    flyVelocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    flyVelocity.Velocity = Vector3.zero
    flyVelocity.Parent = root
    flyGyro = Instance.new("BodyGyro")
    flyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    flyGyro.P = 9e4
    flyGyro.Parent = root
    flyConnection = RunService.RenderStepped:Connect(function()
        local camera = workspace.CurrentCamera
        local direction = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then
            direction += camera.CFrame.LookVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then
            direction -= camera.CFrame.LookVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then
            direction += camera.CFrame.RightVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then
            direction -= camera.CFrame.RightVector
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
            direction += Vector3.yAxis
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
            direction -= Vector3.yAxis
        end
        if direction.Magnitude > 0 then
            flyVelocity.Velocity = direction.Unit * flySpeed
        else
            flyVelocity.Velocity = Vector3.zero
        end
        flyGyro.CFrame = camera.CFrame
    end)
end

local function updateEsp()
    local camera = workspace.CurrentCamera
    local viewport = camera.ViewportSize
    local localCharacter = LocalPlayer.Character
    local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
    for _, player in Players:GetPlayers() do
        if player ~= LocalPlayer then
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            local head = character and character:FindFirstChild("Head")

            local highlight = character and character:FindFirstChild("ESPHighlight")
            if character and espHighlightEnabled then
                if not highlight then
                    highlight = Instance.new("Highlight")
                    highlight.Name = "ESPHighlight"
                    highlight.FillColor = Color3.fromRGB(255, 0, 0)
                    highlight.OutlineColor = WHITE
                    highlight.FillTransparency = 0.5
                    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    highlight.Parent = character
                end
            elseif highlight then
                highlight:Destroy()
            end

            local billboard = head and head:FindFirstChild("ESPName")
            if head and espNamesEnabled then
                if not billboard then
                    billboard = Instance.new("BillboardGui")
                    billboard.Name = "ESPName"
                    billboard.AlwaysOnTop = true
                    billboard.Size = UDim2.fromOffset(200, 30)
                    billboard.StudsOffset = Vector3.new(0, 3, 0)
                    local label = Instance.new("TextLabel")
                    label.Name = "Label"
                    label.Size = UDim2.fromScale(1, 1)
                    label.BackgroundTransparency = 1
                    label.TextColor3 = WHITE
                    label.TextStrokeTransparency = 0
                    label.Font = Enum.Font.GothamBold
                    label.TextSize = 14
                    label.Parent = billboard
                    billboard.Parent = head
                end
                local label = billboard:FindFirstChild("Label")
                if label then
                    local distance = 0
                    if localRoot and root then
                        distance = (localRoot.Position - root.Position).Magnitude
                    end
                    label.Text = player.Name .. " [" .. math.floor(distance) .. "]"
                end
            elseif billboard then
                billboard:Destroy()
            end

            if espTracersEnabled and root and Drawing then
                local line = tracers[player]
                if not line then
                    line = Drawing.new("Line")
                    line.Thickness = 1
                    line.Color = WHITE
                    tracers[player] = line
                end
                local point, onScreen = camera:WorldToViewportPoint(root.Position)
                if onScreen then
                    line.From = Vector2.new(viewport.X / 2, viewport.Y)
                    line.To = Vector2.new(point.X, point.Y)
                    line.Visible = true
                else
                    line.Visible = false
                end
            else
                removeTracer(player)
            end
        end
    end
end

RunService.RenderStepped:Connect(updateEsp)

local function getRole(player)
    local team = player.Team
    if team and (team.Name == "Detective" or team.Name == "Guard" or team.Name == "Player") then
        return team.Name
    end
    if player:GetAttribute("IsDetective") then
        return "Detective"
    end
    if player:GetAttribute("IsGuard") then
        return "Guard"
    end
    return "Player"
end

local function canAimAt(player)
    local myRole = getRole(LocalPlayer)
    local role = getRole(player)
    local isFrontman = not not player:GetAttribute(FRONTMAN_ATTRIBUTE)
    if myRole == "Detective" then
        return role == "Guard"
    elseif myRole == "Guard" then
        return role == "Detective" or (role == "Player" and not isFrontman)
    end
    return role == "Guard" or isFrontman
end

local function getAimTarget(origin)
    local camera = workspace.CurrentCamera
    local look = camera.CFrame.LookVector
    local best
    local bestDot = -1
    for _, player in Players:GetPlayers() do
        if player ~= LocalPlayer and player.Character and not player:GetAttribute("Dead") and canAimAt(player) then
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            local part = player.Character:FindFirstChild("Head") or player.Character:FindFirstChild("HumanoidRootPart")
            if humanoid and humanoid.Health > 0 and part then
                local offset = part.Position - origin
                if offset.Magnitude > 0 then
                    local dot = look:Dot(offset.Unit)
                    if dot > bestDot then
                        bestDot = dot
                        best = part
                    end
                end
            end
        end
    end
    return best
end

local function setAimGun(state)
    local weapon = getBulletWeapon()
    if not weapon then
        return
    end
    if state then
        if not originalFire then
            originalFire = weapon.fire
        end
        weapon.fire = function(self, origin, direction, charge)
            local target = getAimTarget(origin)
            if target then
                direction = (target.Position - origin).Unit
            end
            return originalFire(self, origin, direction, charge)
        end
    elseif originalFire then
        weapon.fire = originalFire
    end
end

local fastBullets = false
local bulletSpeedValue = 5000
local fastFire = false
local fireCooldownValue = 0.05
local damageEnabled = false
local damageMultiplier = 2
local passThrough = false
local noRecoil = false
local bulletPatched = false

local function getTargetCharacters()
    local targets = {}
    for _, player in Players:GetPlayers() do
        if player ~= LocalPlayer and player.Character and not player:GetAttribute("Dead") and canAimAt(player) then
            table.insert(targets, player.Character)
        end
    end
    return targets
end

local function containsTarget(instance, targets)
    for _, character in targets do
        if instance == character or character:IsDescendantOf(instance) then
            return true
        end
    end
    return false
end

local function collectIgnore(parent, targets, list)
    for _, child in parent:GetChildren() do
        if table.find(targets, child) then
            continue
        elseif containsTarget(child, targets) then
            collectIgnore(child, targets, list)
        else
            table.insert(list, child)
        end
    end
end

local function buildIgnoreList()
    local list = {}
    collectIgnore(workspace, getTargetCharacters(), list)
    return list
end

local function patchBullets()
    if bulletPatched then
        return
    end
    local base = getBaseWeapon()
    local bullet = getBulletWeapon()
    if not base or not bullet then
        notify("Armas", "No pude cargar los módulos del sistema de armas")
        return
    end
    bulletPatched = true

    local baseGetConfigValue = base.getConfigValue
    base.getConfigValue = function(self, name, default)
        if fastBullets and name == "BulletSpeed" then
            return bulletSpeedValue
        end
        if fastFire then
            if name == "ShotCooldown" or name == "BurstShotCooldown" then
                return fireCooldownValue
            end
            if name == "FireMode" then
                return "Automatic"
            end
        end
        if damageEnabled and name == "HitDamage" then
            local value = baseGetConfigValue(self, name, default)
            return (tonumber(value) or 10) * damageMultiplier
        end
        if noRecoil and string.find(string.lower(name), "recoil", 1, true) then
            local value = baseGetConfigValue(self, name, default)
            if type(value) == "number" then
                return 0
            end
            return value
        end
        return baseGetConfigValue(self, name, default)
    end

    local baseGetIgnoreList = bullet.getIgnoreList
    for _, methodName in { "applyRecoil", "addRecoil", "recoil" } do
        local method = base[methodName] or bullet[methodName]
        if type(method) == "function" then
            local owner = base[methodName] and base or bullet
            owner[methodName] = function(self, ...)
                if noRecoil then
                    return
                end
                return method(self, ...)
            end
        end
    end
    bullet.getIgnoreList = function(self, isLocal)
        local list = baseGetIgnoreList(self, isLocal)
        if passThrough then
            list = buildIgnoreList()
            self.ignoreList = list
        end
        return list
    end

    local baseOnHit = bullet.onHit
    bullet.onHit = function(self, hitInfo)
        baseOnHit(self, hitInfo)
        if damageEnabled and damageMultiplier > 1 and self.player == LocalPlayer then
            local target = hitInfo.h
            if typeof(target) == "Instance" and target:IsA("Humanoid") then
                local remote = self.weaponsSystem.getRemoteEvent("WeaponHit")
                for _ = 2, damageMultiplier do
                    local copy = {}
                    for key, value in hitInfo do
                        copy[key] = value
                    end
                    remote:FireServer(self.instance, copy)
                end
            end
        end
    end
end

local killHidersEnabled = false
local killSeekersEnabled = false
local knifeFastEnabled = false
local attackInterval = 0.1
local lastAttack = 0
local cooldownRestores = {}
local boostedTools = setmetatable({}, { __mode = "k" })

local function nameMatches(name, patterns)
    local lower = string.lower(name)
    for _, pattern in patterns do
        if string.find(lower, pattern, 1, true) then
            return true
        end
    end
    return false
end

local function isCooldownName(name)
    return nameMatches(name, { "cooldown", "delay", "debounce" })
end

local function boostCooldowns(tool)
    if boostedTools[tool] then
        return
    end
    boostedTools[tool] = true
    for name, value in tool:GetAttributes() do
        if type(value) == "number" and isCooldownName(name) then
            table.insert(cooldownRestores, function()
                if tool.Parent then
                    tool:SetAttribute(name, value)
                end
            end)
            tool:SetAttribute(name, 0.05)
        end
    end
    for _, descendant in tool:GetDescendants() do
        if (descendant:IsA("NumberValue") or descendant:IsA("IntValue")) and isCooldownName(descendant.Name) then
            local value = descendant.Value
            table.insert(cooldownRestores, function()
                if descendant.Parent then
                    descendant.Value = value
                end
            end)
            descendant.Value = 0.05
        end
    end
end

local function restoreCooldowns()
    for _, restore in cooldownRestores do
        restore()
    end
    table.clear(cooldownRestores)
    table.clear(boostedTools)
end

        end
    end
end

local PUSH_TOOL_NAMES = { "push", "empuj" }
local pushBoostEnabled = false
local pushPower = 300
local pushRange = 8
local pushDuration = 0.35
local pushActive = false
local pushToolConnections = setmetatable({}, { __mode = "k" })

local function getPushTarget(root)
    local best
    local bestScore = math.huge
    local look = root.CFrame.LookVector
    for _, player in Players:GetPlayers() do
        if player ~= LocalPlayer and not player:GetAttribute("Dead") then
            local character = player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local targetRoot = character and character:FindFirstChild("HumanoidRootPart")
            if humanoid and targetRoot and humanoid.Health > 0 then
                local offset = targetRoot.Position - root.Position
                local distance = offset.Magnitude
                if distance > 0 and distance <= pushRange then
                    local facing = look:Dot(offset.Unit)
                    if facing > 0.2 then
                        local score = distance * (2 - facing)
                        if score < bestScore then
                            bestScore = score
                            best = targetRoot
                        end
                    end
                end
            end
        end
    end
    return best
end

local function doStrongPush()
    if pushActive then
        return
    end
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then
        return
    end
    local target = getPushTarget(root)
    if not target then
        return
    end
    pushActive = true
    local startPosition = root.CFrame
    local deadline = os.clock() + pushDuration
    local heartbeatConnection
    local steppedConnection
    local function finish()
        if heartbeatConnection then
            heartbeatConnection:Disconnect()
        end
        if steppedConnection then
            steppedConnection:Disconnect()
        end
        if root.Parent then
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            if (root.Position - startPosition.Position).Magnitude > pushRange * 2 then
                root.CFrame = startPosition
            end
        end
        pushActive = false
    end
    local saved = root.AssemblyLinearVelocity
    heartbeatConnection = RunService.Heartbeat:Connect(function()
        if os.clock() > deadline or not root.Parent or not target.Parent then
            finish()
            return
        end
        local offset = target.Position - root.Position
        local flat = Vector3.new(offset.X, 0, offset.Z)
        local direction = flat.Magnitude > 0.01 and flat.Unit or root.CFrame.LookVector
        saved = root.AssemblyLinearVelocity
        root.AssemblyLinearVelocity = direction * pushPower + Vector3.new(0, pushPower * 0.35, 0)
    end)
    steppedConnection = RunService.Stepped:Connect(function()
        if root.Parent then
            root.AssemblyLinearVelocity = saved
        end
    end)
end

local function hookPushTool(tool)
    if pushToolConnections[tool] or not tool:IsA("Tool") or not nameMatches(tool.Name, PUSH_TOOL_NAMES) then
        return
    end
    pushToolConnections[tool] = tool.Activated:Connect(function()
        if pushBoostEnabled then
            task.spawn(doStrongPush)
        end
    end)
end

local function hookPushContainers()
    local character = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    for _, container in { character, backpack } do
        if container then
            for _, child in container:GetChildren() do
                hookPushTool(child)
            end
        end

local LIGHTS_WEAPON_NAMES = { "fist", "punch", "puño", "puno", "knife", "cuchillo", "sword", "bat", "bate", "pipe", "tubo", "bottle", "botella", "stick", "palo", "hammer", "martillo", "axe", "hacha", "weapon", "arma" }
local lightsKillTargetName
local lightsKillRunning = false
local lightsKillInterval = 0.1
local lightsKillTimeout = 20

local function findLightsWeapon()
    local tool = findTool(LIGHTS_WEAPON_NAMES)
    if tool then
        return tool
    end
    local character = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    for _, container in { character, backpack } do
        if container then
            for _, child in container:GetChildren() do

        end

end

local function isTargetAlive(player)
    if not player or player.Parent ~= Players or player:GetAttribute("Dead") then
        return false
    end
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local targetRoot = character and character:FindFirstChild("HumanoidRootPart")
    return humanoid and targetRoot and humanoid.Health > 0, targetRoot
end

local function stopLightsKill()
    lightsKillRunning = false
end

local function startLightsKill()
    if lightsKillRunning then
        return
    end
    local target = lightsKillTargetName and Players:FindFirstChild(lightsKillTargetName)
    if not target or target == LocalPlayer then
        notify("Lights Out", "Elige un jugador de la lista")
        return
    end
    if not isTargetAlive(target) then
        notify("Lights Out", target.Name .. " no está vivo")
        return
    end
    local weapon = findLightsWeapon()
    if not weapon then
        notify("Lights Out", "No tienes ninguna tool de daño (puños, arma...) en la mochila")
        return
    end
    lightsKillRunning = true
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local returnCFrame = root and root.CFrame
    boostCooldowns(weapon)
    notify("Lights Out", "Atacando a " .. target.Name .. " con " .. weapon.Name)
    local deadline = os.clock() + lightsKillTimeout
    local killed = false
    while lightsKillRunning and os.clock() < deadline do
        local alive, targetRoot = isTargetAlive(target)
        if not alive then
            killed = true
            break
        end
        character = LocalPlayer.Character
        root = character and character:FindFirstChild("HumanoidRootPart")
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not root or not humanoid or humanoid.Health <= 0 then
            break
        end
        if not weapon.Parent or (weapon.Parent ~= character and weapon.Parent ~= LocalPlayer:FindFirstChildOfClass("Backpack")) then
            weapon = findLightsWeapon()
            if not weapon then
                break
            end
            boostCooldowns(weapon)
        end
        if weapon.Parent ~= character then
            humanoid:EquipTool(weapon)
        end
        root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 2.5)
        root.AssemblyLinearVelocity = Vector3.zero
        pcall(function()
            weapon:Activate()
        end)
        task.wait(lightsKillInterval)
    end
    lightsKillRunning = false
    if not knifeFastEnabled and not killSeekersEnabled then
        restoreCooldowns()
    end
    local finalRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if returnCFrame and finalRoot then
        finalRoot.CFrame = returnCFrame
        finalRoot.AssemblyLinearVelocity = Vector3.zero
    end
    if killed then
        notify("Lights Out", target.Name .. " eliminado")
    else
        notify("Lights Out", "Detenido sin eliminar a " .. target.Name)
    end
end

local function getNearestEnemy(team, root)
    local closest
    local shortest = math.huge
    for _, player in Players:GetPlayers() do
        if player ~= LocalPlayer
            and player:GetAttribute("HideNSeek_Team") == team
            and not player:GetAttribute("Dead")
            and not player:GetAttribute("IsGuard")
        then
            local character = player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local targetRoot = character and character:FindFirstChild("HumanoidRootPart")
            if humanoid and targetRoot and humanoid.Health > 0 then
                local distance = (targetRoot.Position - root.Position).Magnitude
                if distance < shortest then
                    shortest = distance
                    closest = targetRoot
                end
            end
        end
    end
    return closest
end

local function performAttack(patterns, targetTeam, boost)
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then
        return false
    end
    local tool = findTool(patterns)
    if not tool then
        return false
    end
    if boost then
        boostCooldowns(tool)
    end
    if targetTeam then
        local target = getNearestEnemy(targetTeam, root)
        if not target then
            return false
        end
        if tool.Parent ~= character then
            humanoid:EquipTool(tool)
        end
        root.CFrame = target.CFrame * CFrame.new(0, 0, 2.5)
        tool:Activate()
        return true
    end
    if tool.Parent ~= character then
        return false
    end
    tool:Activate()
    return true
end

local function setFootsteps(state)
    footstepsEnabled = state
    local class = getHideNSeekClass()
    if not class then
        return
    end
    if not footstepsPatched then
        footstepsPatched = true
        local baseHandle = class.handle
        local baseDisable = class.disablePlayerFootsteps
        class.handle = function(self)
            baseHandle(self)
            if footstepsEnabled then
                self:enablePlayerFootsteps()
            end
        end
        class.disablePlayerFootsteps = function(self)
            if footstepsEnabled then
                return
            end
            return baseDisable(self)
        end
    end
    if state and getgc then
        pcall(function()
            for _, value in getgc(true) do
                if type(value) == "table" and rawget(value, "footstepsMaid") and rawget(value, "map") then
                    value:enablePlayerFootsteps()
                    break
                end
            end
        end)
    end
end

local glassReplies = 0
local glassDeadlyCount = 0

local function markGlass(part)
    if not part:IsA("BasePart") or not glassEnabled then
        return
    end
    local remote = getGlassPass()
    if not remote then
        return
    end
    local ok, deadly = pcall(function()
        return remote:InvokeServer(part.Name)
    end)
    if not ok or not glassEnabled or not part.Parent then
        return
    end
    glassReplies += 1
    if deadly then
        glassDeadlyCount += 1
        if not glassOriginal[part] then
            glassOriginal[part] = { Color = part.Color, Transparency = part.Transparency }
        end
        part.Color = GLASS_DEADLY
        part.Transparency = 0.45
    end
    if glassReplies == 20 and glassDeadlyCount == 0 then
        notify("Glass", "El servidor no marca ninguno como mortal")
    end
end

local honeycombEnabled = false
local honeycombRunning = false
local honeycombBatch = 4
local honeycombInterval = 0.2

local function runHoneycomb(force)
    if honeycombRunning then
        return
    end
    honeycombRunning = true
    local ok, err = pcall(function()
        local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
        local gui = playerGui and playerGui:WaitForChild("Honeycomb", 10)
        local shapeValue = gui and gui:WaitForChild("Shape", 10)
        local connection = gui and gui:WaitForChild("Connection", 10)
        if not shapeValue or not connection then
            return
        end
        local cookie = shapeValue.Value
        local path = cookie and cookie:FindFirstChild("Path")
        if not path then
            notify("Honeycomb", "No encontré la galleta")
            return
        end
        task.wait(1)
        local choose = gui:FindFirstChild("ChooseInputType")
        while choose and choose.Parent and choose.Visible do
            task.wait(0.2)
        end
        local pending = {}
        for _, part in path:GetChildren() do
            if part:IsA("BasePart") and not part:GetAttribute("Completed") then
                table.insert(pending, part)
            end
        end
        local index = 1
        while (force or honeycombEnabled)
            and LocalPlayer:GetAttribute("HONEYCOMB_START_CUTTING")
            and index <= #pending
        do
            local batch = {}
            for i = index, math.min(index + honeycombBatch - 1, #pending) do
                table.insert(batch, pending[i])
            end
            connection:FireServer(batch)
            for _, part in batch do
                part.Transparency = 0
            end
            index += honeycombBatch
            task.wait(honeycombInterval)
        end
        if index > #pending then
            notify("Honeycomb", "Puntos enviados")
        end
    end)
    honeycombRunning = false
    if not ok then
        warn(err)
    end
end

LocalPlayer:GetAttributeChangedSignal("HONEYCOMB_START_CUTTING"):Connect(function()
    if honeycombEnabled and LocalPlayer:GetAttribute("HONEYCOMB_START_CUTTING") then
        task.spawn(runHoneycomb, false)
    end
end)
