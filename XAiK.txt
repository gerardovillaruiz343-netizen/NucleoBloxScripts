local function fmt(v)
    local t = typeof(v)
    if t == "Instance" then return v:GetFullName() end
    if t == "Vector3" then return string.format("Vector3(%.2f, %.2f, %.2f)", v.X, v.Y, v.Z) end
    if t == "Color3" then return string.format("Color3(%d, %d, %d)", math.floor(v.R * 255), math.floor(v.G * 255), math.floor(v.B * 255)) end
    if t == "string" then return '"' .. v:sub(1, 120) .. '"' end
    return tostring(v)
end

local function getProps(inst)
    local out = {}
    for _, p in ipairs(knownProps) do
        local ok, val = pcall(function() return inst[p] end)
        if ok and val ~= nil and typeof(val) ~= "RBXScriptSignal" and typeof(val) ~= "function" then
            table.insert(out, p .. "=" .. fmt(val))
        end
    end
    pcall(function()
        for k, v in pairs(inst:GetAttributes()) do
            table.insert(out, "@" .. k .. "=" .. fmt(v))
        end
    end)
    return out
end

local function resolvePath(path)
    path = path:gsub("^%s+", ""):gsub("%s+$", ""):gsub("/", ".")
    local parts = {}
    for seg in path:gmatch("[^%.]+") do table.insert(parts, seg) end
    local cur = game
    local i = 1
    if parts[1] == "game" then i = 2 end
    local first = parts[i]
    if first == "LocalPlayer" then
        cur = LocalPlayer
        i = i + 1
    elseif first == "Character" then
        cur = LocalPlayer.Character
        i = i + 1
    elseif first == "workspace" then
        cur = workspace
        i = i + 1
    end
    while cur and i <= #parts do
        local name = parts[i]
        local nxt = cur:FindFirstChild(name)
        if not nxt and cur == game then
            pcall(function() nxt = game:GetService(name) end)
        end
        cur = nxt
        i = i + 1
    end
    return cur
end

local perfPresets = {
    Bajo = { nodes = 12000, decompile = 25, budget = 0.004, maxTools = 4, cacheChars = 600000 },
    Medio = { nodes = 35000, decompile = 70, budget = 0.007, maxTools = 6, cacheChars = 1500000 },

}

local function perf()
    return perfPresets[config.perf] or perfPresets.Bajo
end

local lastYield = os.clock()
local function breathe()
    if os.clock() - lastYield > perf().budget then
        task.wait()
        lastYield = os.clock()
    end
end

local skipServices = {
    CoreGui = true, CorePackages = true, RobloxReplicatedStorage = true, RobloxPluginGuiService = true,
    PluginGuiService = true, PluginDebugService = true, TestService = true, Chat = true,
}

local function isCoreInstance(inst)
    local ok, full = pcall(function() return inst:GetFullName() end)
    if not ok then return true end
    local top = full:match("^([^%.]+)")
    return top ~= nil and skipServices[top] == true
end

local function walk(root, visit, maxNodes)
    maxNodes = maxNodes or perf().nodes
    local stack = { root }
    local count = 0
    while #stack > 0 do
        local inst = table.remove(stack)
        local ok, children = pcall(function() return inst:GetChildren() end)
        if ok then
            for _, c in ipairs(children) do
                local skip = inst == game and skipServices[c.Name]
                if not skip then
                    count = count + 1
                    if count > maxNodes then return count, true end
                    if visit(c) then return count, false end
                    stack[#stack + 1] = c
                    breathe()
                end
            end
        end
    end
    return count, false
end

local function inspect(path)
    local inst = resolvePath(path)
    if not inst then return "No existe: " .. path end
    local lines = { inst:GetFullName() .. " [" .. inst.ClassName .. "]" }
    local props = getProps(inst)
    if #props > 0 then table.insert(lines, "Propiedades: " .. table.concat(props, ", ")) end
    if inst:IsA("LuaSourceContainer") then
        local src
        pcall(function() src = decompile and decompile(inst) end)
        table.insert(lines, src and ("Código:\n" .. src:sub(1, 3000)) or "Código: (decompile no disponible)")
    end
    local kids = inst:GetChildren()
    if #kids > 0 then
        table.insert(lines, "Hijos (" .. #kids .. "):")
        for idx, c in ipairs(kids) do
            if idx > 40 then table.insert(lines, "  ...") break end
            local cp = getProps(c)
            table.insert(lines, "  " .. c.Name .. " [" .. c.ClassName .. "]" .. (#cp > 0 and (" " .. table.concat(cp, ", "):sub(1, 150)) or ""))
        end
    end
    return table.concat(lines, "\n")
end

local function searchGame(query)
    local terms = {}
    local q = query:lower()
    table.insert(terms, q)
    for word in q:gmatch("[%wáéíóúñ]+") do
        if keywordMap[word] then
            for _, t in ipairs(keywordMap[word]) do table.insert(terms, t:lower()) end
        end
    end
    local matches, seen = {}, {}
    local function add(inst, reason)
        if seen[inst] or #matches >= 50 then return end
        seen[inst] = true
        local props = getProps(inst)
        table.insert(matches, inst:GetFullName() .. " [" .. inst.ClassName .. "] (" .. reason .. ")" .. (#props > 0 and (" " .. table.concat(props, ", "):sub(1, 200)) or ""))
    end
    local function check(inst)
        local name = inst.Name:lower()
        local cls = inst.ClassName:lower()
        for _, t in ipairs(terms) do
            if name:find(t, 1, true) then add(inst, "nombre") return end
            if cls:find(t, 1, true) then add(inst, "clase") return end
            for _, p in ipairs(knownProps) do
                if p:lower() == t then
                    local ok, v = pcall(function() return inst[p] end)
                    if ok and v ~= nil then add(inst, "propiedad " .. p) return end
                end
            end
            local okA, attrs = pcall(function() return inst:GetAttributes() end)
            if okA then
                for k in pairs(attrs) do
                    if k:lower():find(t, 1, true) then add(inst, "atributo " .. k) return end
                end
            end
        end
    end
    local roots = { workspace, game:GetService("ReplicatedStorage"), game:GetService("ReplicatedFirst"), game:GetService("Lighting"), game:GetService("StarterGui"), game:GetService("StarterPack"), game:GetService("StarterPlayer"), Players }
    local budget = perf().nodes
    local truncated = false
    for _, root in ipairs(roots) do
        if #matches >= 50 or budget <= 0 then break end
        local used, cut = walk(root, function(d)
            pcall(check, d)
            return #matches >= 50
        end, budget)
        budget = budget - used
        truncated = truncated or cut
    end
    local note = (truncated or budget <= 0) and "\n(Búsqueda recortada por el límite de rendimiento.)" or ""
    if #matches == 0 then return "Sin resultados para: " .. query .. " (términos probados: " .. table.concat(terms, ", ") .. ")" .. note end
    return table.concat(matches, "\n") .. note
end

local scriptListCache, scriptListTime = nil, 0

local function allScripts()
    if scriptListCache and os.clock() - scriptListTime < 30 then return scriptListCache end
    local list, seen = {}, {}
    local function add(s)
        if typeof(s) ~= "Instance" or seen[s] then return end
        seen[s] = true
        local ok, isScript = pcall(function() return s:IsA("LuaSourceContainer") end)
        if ok and isScript and not isCoreInstance(s) then
            table.insert(list, s)
        end
    end
    local function addList(getter)
        local ok, items = pcall(getter)
        if ok and type(items) == "table" then
            for _, s in ipairs(items) do
                add(s)
                breathe()
            end
        end
    end
    if getscripts then addList(getscripts) end
    if getloadedmodules then addList(getloadedmodules) end
    walk(game, function(d)
        add(d)
        return #list >= 3000
    end)
    if getnilinstances then addList(getnilinstances) end
    scriptListCache, scriptListTime = list, os.clock()
    return list
end

local function listScripts(filter)
    local f = filter and filter ~= "" and filter:lower() or nil
    local out = {}
    for _, s in ipairs(allScripts()) do
        local full = s:GetFullName()
        if not f or full:lower():find(f, 1, true) then
            table.insert(out, full .. " [" .. s.ClassName .. "]")
            if #out >= 200 then table.insert(out, "...") break end

    if #out == 0 then return "No se encontraron scripts" .. (f and (" con: " .. filter) or "") .. "." .. note end
    return table.concat(out, "\n") .. note
end

local sourceCache = {}
local sourceOrder = {}
local sourceChars = 0

local function cacheSource(s, src)
    sourceCache[s] = src
    table.insert(sourceOrder, s)
    sourceChars = sourceChars + #src
    while sourceChars > perf().cacheChars and #sourceOrder > 1 do
        local old = table.remove(sourceOrder, 1)
        local oldSrc = sourceCache[old]
        if oldSrc then sourceChars = sourceChars - #oldSrc end
        sourceCache[old] = nil
    end
end

local function getSource(s, noCache)
    if sourceCache[s] then return sourceCache[s] end
    local src
    if decompile then
        local ok, r = pcall(decompile, s)
        if ok and type(r) == "string" and r:find("%S") then src = r end
        task.wait()
        lastYield = os.clock()
    end
    if not src then
        pcall(function()
            local r = s.Source
            if type(r) == "string" and r:find("%S") then src = r end
        end)
    end
    if src and not noCache then cacheSource(s, src) end
    return src
end

local function searchCode(query)
    if not decompile then return "Este executor no tiene decompile; no se puede leer el código de los scripts. Usa [[scripts]] para ver nombres." end
    local q = query:lower()
    local out = {}
    local scanned, failed = 0, 0
    local limit = perf().decompile
    local scripts = allScripts()
    for _, s in ipairs(scripts) do
        if scanned + failed >= limit or #out >= 40 then break end
        local src = getSource(s, true)
        if src then
            scanned = scanned + 1
            local ln = 0
            for line in (src .. "\n"):gmatch("(.-)\n") do
                ln = ln + 1
                if line:lower():find(q, 1, true) then
                    table.insert(out, s:GetFullName() .. " [" .. s.ClassName .. "] línea " .. ln .. ": " .. line:gsub("^%s+", ""):sub(1, 160))
                    if #out >= 40 then break end
                end
                if ln % 400 == 0 then breathe() end
            end
        else
            failed = failed + 1
        end
    end
    local summary = "\n(Scripts revisados: " .. scanned .. " de " .. #scripts .. ", sin poder leer: " .. failed .. ")"
    if scanned + failed >= limit and #scripts > limit then
        summary = summary .. "\n(Límite de rendimiento alcanzado. Para revisar menos scripts pero más precisos usa [[scripts: filtro]] y luego [[codigo: ruta]].)"
    end
    if #out == 0 then return "No se encontró \"" .. query .. "\" en el código." .. summary end
    return table.concat(out, "\n") .. summary
end

local function dumpScripts(filter)
    if not decompile then return "Este executor no tiene decompile; no se puede extraer el código." end
    if not writefile then return "Este executor no permite guardar archivos." end
    local f = filter and filter ~= "" and filter:lower() or nil
    local base = "XAiK/Dumps/" .. tostring(game.PlaceId)
    local okFolder = pcall(function()
        if not isfolder("XAiK") then makefolder("XAiK") end
        if not isfolder("XAiK/Dumps") then makefolder("XAiK/Dumps") end
        if not isfolder(base) then makefolder(base) end
    end)
    if not okFolder then return "No se pudo crear la carpeta " .. base end
    local saved, failed, names = 0, 0, {}
    local maxDump = perf().decompile * 4
    for _, s in ipairs(allScripts()) do
        if saved + failed >= maxDump then break end
        local full = s:GetFullName()
        if not f or full:lower():find(f, 1, true) then
            local src = getSource(s, true)
            task.wait()
            if src then
                local fileName = full:gsub("[^%w%._]", "_") .. ".lua"
                if pcall(writefile, base .. "/" .. fileName, src) then
                    saved = saved + 1
                    if #names < 40 then table.insert(names, fileName) end
                else
                    failed = failed + 1
                end
            else
                failed = failed + 1
            end
        end
    end
    return "Guardados " .. saved .. " scripts en workspace del executor/" .. base .. " (fallidos: " .. failed .. ")\n" .. table.concat(names, "\n") .. (saved > #names and "\n..." or "")
end

function Control.findPlayer(name)
    local n = tostring(name):lower():gsub("^@", "")
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():sub(1, #n) == n or p.DisplayName:lower():sub(1, #n) == n then return p end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():find(n, 1, true) or p.DisplayName:lower():find(n, 1, true) then return p end
    end
    return nil
end

function Control.players()
    local out = {}
    local _, _, root = charParts()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local dist = ""
            local r = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if root and r then dist = string.format(" (%.0f studs)", (r.Position - root.Position).Magnitude) end
            table.insert(out, p.DisplayName .. " @" .. p.Name .. dist)
        end
    end
    return out
end

local function toPosition(target)
    local t = typeof(target)
    if t == "Vector3" then return target end
    if t == "CFrame" then return target.Position end
    if t == "Instance" then
        if target:IsA("Player") then
            local r = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
            return r and r.Position
        end
        if target:IsA("BasePart") then return target.Position end
        if target:IsA("Model") then return target:GetPivot().Position end
        if target:IsA("Attachment") then return target.WorldPosition end
        local part = target:FindFirstChildWhichIsA("BasePart", true)
        return part and part.Position
    end
    if t == "string" then
        local p = Control.findPlayer(target)
        if p then return toPosition(p) end
        local inst = resolvePath(target)
        if inst then return toPosition(inst) end
    end
    return nil
end
Control.position = toPosition

function Control.teleport(target)
    local _, _, root = charParts()
    local pos = toPosition(target)
    if not root then return false, "No hay personaje" end
    if not pos then return false, "Destino no encontrado" end
    root.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
    return true
end

function Control.walkTo(target, timeout)
    local _, hum, root = charParts()
    local pos = toPosition(target)
    if not hum or not root then return false, "No hay personaje" end
    if not pos then return false, "Destino no encontrado" end
    local path = PathfindingService:CreatePath({ AgentCanJump = true })
    local ok = pcall(function() path:ComputeAsync(root.Position, pos) end)
    local waypoints
    if ok and path.Status == Enum.PathStatus.Success then
        waypoints = path:GetWaypoints()
    else
        waypoints = { { Position = pos, Action = Enum.PathWaypointAction.Walk } }
    end
    local deadline = os.clock() + (timeout or 30)
    for _, wp in ipairs(waypoints) do
        if wp.Action == Enum.PathWaypointAction.Jump then hum.Jump = true end
        hum:MoveTo(wp.Position)
        local reached = false
        local conn = hum.MoveToFinished:Connect(function() reached = true end)
        local t0 = os.clock()
        while not reached and os.clock() - t0 < 4 and os.clock() < deadline do task.wait() end
        conn:Disconnect()
        if os.clock() >= deadline then return false, "Tiempo agotado" end
    end
    return true
end

local walkDirs = {
    adelante = Vector3.new(0, 0, -1), forward = Vector3.new(0, 0, -1),
    atras = Vector3.new(0, 0, 1), ["atrás"] = Vector3.new(0, 0, 1), back = Vector3.new(0, 0, 1),
    izquierda = Vector3.new(-1, 0, 0), left = Vector3.new(-1, 0, 0),
    derecha = Vector3.new(1, 0, 0), right = Vector3.new(1, 0, 0),
}

function Control.walk(direction, seconds)
    local _, hum = charParts()
    if not hum then return false, "No hay personaje" end
    local v = typeof(direction) == "Vector3" and direction or walkDirs[tostring(direction):lower()] or walkDirs.adelante
    local t0 = os.clock()
    while os.clock() - t0 < (seconds or 1) do
        hum:Move(v, true)
        RunService.Heartbeat:Wait()
    end
    hum:Move(Vector3.new(0, 0, 0), true)
    return true
end

function Control.jump()
    local _, hum = charParts()
    if not hum then return false end
    hum.Jump = true
    pcall(function() hum:ChangeState(Enum.HumanoidStateType.Jumping) end)
    return true
end

    hum.Sit = state ~= false
    return true
end

function Control.stopFollow()
    if followConn then
        followConn:Disconnect()
        followConn = nil
    end
    return true
end

function Control.stop()
    Control.stopFollow()
    local _, hum, root = charParts()
    if hum and root then
        hum:Move(Vector3.new(0, 0, 0), true)
        hum:MoveTo(root.Position)
    end
    return true
end

function Control.follow(target, distance)
    Control.stopFollow()
    distance = distance or 5
    followConn = RunService.Heartbeat:Connect(function()
        local _, hum, root = charParts()
        local pos = toPosition(target)
        if hum and root and pos and (pos - root.Position).Magnitude > distance then
            hum:MoveTo(pos)
        end
    end)
    return true
end

function Control.interact(target)
    local inst = typeof(target) == "Instance" and target or resolvePath(tostring(target))
    if not inst then return false, "No existe" end
    local prompt = inst:IsA("ProximityPrompt") and inst or inst:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt then
        if fireproximityprompt then
            pcall(fireproximityprompt, prompt)
        else
            prompt:InputHoldBegin()
            task.wait(prompt.HoldDuration + 0.05)
            prompt:InputHoldEnd()
        end
        return true, "ProximityPrompt"
    end
    local click = inst:IsA("ClickDetector") and inst or inst:FindFirstChildWhichIsA("ClickDetector", true)
    if click and fireclickdetector then
        pcall(fireclickdetector, click)
        return true, "ClickDetector"
    end
    local _, _, root = charParts()
    local part = inst:IsA("BasePart") and inst or inst:FindFirstChildWhichIsA("BasePart", true)
    if part and root and firetouchinterest then
        firetouchinterest(root, part, 0)
        task.wait()
        firetouchinterest(root, part, 1)
        return true, "Touch"
    end
    return false, "No hay forma de interactuar con eso"
end

function Control.chat(msg)
    msg = tostring(msg)
    local ok, err = pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            local channels = TextChatService:FindFirstChild("TextChannels")
            local ch = channels and channels:FindFirstChild("RBXGeneral")
            if not ch and channels then
                for _, c in ipairs(channels:GetChildren()) do
                    if c:IsA("TextChannel") then ch = c break end
                end
            end
            if not ch then error("No hay canal de chat") end
            ch:SendAsync(msg)
        else
            game:GetService("ReplicatedStorage").DefaultChatSystemChatEvents.SayMessageRequest:FireServer(msg, "All")
        end
    end)
    return ok, ok and "Enviado" or tostring(err)
end

local function runCode(code)
    local fn, compileErr = loadstring(code)
    if not fn then return false, "Error de compilación: " .. tostring(compileErr) end
    local output = {}
    local env = setmetatable({
        print = function(...)
            local t = {}
            for i = 1, select("#", ...) do t[i] = tostring(select(i, ...)) end
            table.insert(output, table.concat(t, " "))
        end,
        warn = function(...)
            local t = {}
            for i = 1, select("#", ...) do t[i] = tostring(select(i, ...)) end
            table.insert(output, "WARN: " .. table.concat(t, " "))
        end,
        XAiK = Control,
    }, { __index = getfenv() })
    pcall(setfenv, fn, env)
    local results
    task.spawn(function()
        results = table.pack(pcall(fn))
    end)
    local waited = 0
    while not results and waited < 3 do
        waited = waited + task.wait()
    end
    if not results then
        local msg = "Ejecutándose en segundo plano (el script sigue activo con bucles o esperas)."
        if #output > 0 then msg = msg .. "\nSalida:\n" .. table.concat(output, "\n") end
        return true, msg
    end
    if not results[1] then
        return false, "Error en ejecución: " .. tostring(results[2]) .. (#output > 0 and ("\nSalida:\n" .. table.concat(output, "\n")) or "")
    end
    local ret = {}
    for i = 2, results.n do table.insert(ret, fmt(results[i])) end
    local msg = "Ejecutado correctamente."
    if #output > 0 then msg = msg .. "\nSalida:\n" .. table.concat(output, "\n") end
    if #ret > 0 then msg = msg .. "\nRetorno: " .. table.concat(ret, ", ") end
    return true, msg
end

local function getRequest()
    return (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
end

local function callAPI(content)
    if config.apiKey == "" then
        return nil, "Falta la Api-Key. Ve a la pestaña XAiK Configuration."
    end
    local req = getRequest()
    if not req then return nil, "Tu executor no tiene función request." end

    local provider = providers[config.provider] or providers.Claude
    local model = (config.model ~= "" and config.model) or provider.model
    table.insert(history, { role = "user", content = content })
    while #history > 40 do
        table.remove(history, 1)
        table.remove(history, 1)
    end
    return sendRequest(req, provider, model, false)
end

local function findNvidiaModel(req, exclude)
    local ok, res = pcall(req, { Url = "https://integrate.api.nvidia.com/v1/models", Method = "GET", Headers = { ["Authorization"] = "Bearer " .. config.apiKey } })
    if not ok or not res then return nil end
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, res.Body or res.body or "")
    if not ok2 or type(data) ~= "table" or type(data.data) ~= "table" then return nil end
    local available = {}
    for _, m in ipairs(data.data) do available[m.id] = true end
    for _, id in ipairs(nvidiaPreferred) do
        if available[id] and id ~= exclude then return id end
    end
    local skip = { "embed", "guard", "safety", "clip", "detector", "riva", "gliner", "calibration", "retriever", "starcoder", "cosmos" }
    for _, m in ipairs(data.data) do
        local bad = m.id == exclude
        for _, s in ipairs(skip) do
            if m.id:lower():find(s, 1, true) then bad = true break end
        end
        if not bad then return m.id end
    end
    return nil
end

function sendRequest(req, provider, model, retried, maxTokens)
    maxTokens = maxTokens or 4096
    local url, body, headers
    local systemPrompt = baseSystemPrompt
    local custom = tostring(config.customInstructions or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if custom ~= "" then
        systemPrompt = systemPrompt .. "\n\nINSTRUCCIONES PERSONALIZADAS DEL USUARIO (síguelas siempre, tienen prioridad sobre tu estilo por defecto, pero no sobre el formato de las herramientas):\n" .. custom
    end
    if config.provider == "Claude" then
        url = provider.url
        body = HttpService:JSONEncode({ model = model, max_tokens = maxTokens, system = systemPrompt, messages = history })
        headers = { ["Content-Type"] = "application/json", ["x-api-key"] = config.apiKey, ["anthropic-version"] = "2023-06-01" }
    elseif config.provider == "Google" then
        local contents = {}
        for _, m in ipairs(history) do
            table.insert(contents, { role = m.role == "assistant" and "model" or "user", parts = { { text = m.content } } })
        end
        url = provider.url:gsub("gemini%-2%.0%-flash", model) .. "?key=" .. config.apiKey
        body = HttpService:JSONEncode({ contents = contents, systemInstruction = { parts = { { text = systemPrompt } } }, generationConfig = { maxOutputTokens = maxTokens } })
        headers = { ["Content-Type"] = "application/json" }
    else
        local msgs = { { role = "system", content = systemPrompt } }
        for _, m in ipairs(history) do table.insert(msgs, m) end
        url = provider.url
        local payload = { model = model, messages = msgs, max_tokens = maxTokens }
        if config.provider == "Groq" and model:lower():find("qwen", 1, true) then
            payload.reasoning_format = "hidden"
        end
        body = HttpService:JSONEncode(payload)
        headers = { ["Content-Type"] = "application/json", ["Authorization"] = "Bearer " .. config.apiKey }
    end

    local ok, response = pcall(req, { Url = url, Method = "POST", Headers = headers, Body = body })
    if not ok or not response then
        table.remove(history)
        return nil, "Error de red: " .. tostring(response)
    end

    local raw = response.Body or response.body or ""
    local status = tonumber(response.StatusCode or response.statusCode or response.Status) or 0

    if config.provider == "Nvidia" and (status == 404 or status == 410) and not retried then
        local newModel = findNvidiaModel(req, model)
        if newModel then
            config.model = newModel
            saveConfig()
            pcall(function() ModelInput:Set(newModel) end)
            WindUI:Notify({ Title = "XAiK", Content = "Modelo no disponible (" .. model .. "). Usando: " .. newModel, Icon = "refresh-cw", Duration = 6 })
            return sendRequest(req, provider, newModel, true, maxTokens)
        end
    end

    local ok2, parsed = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok2 or type(parsed) ~= "table" then
        table.remove(history)
        return nil, "HTTP " .. status .. " (" .. config.provider .. ", modelo " .. model .. "): " .. raw:sub(1, 200)
    end

    local text, finish, reasoning
    pcall(function()
        local parts = {}
        if config.provider == "Claude" then
            finish = parsed.stop_reason
            for _, b in ipairs(parsed.content or {}) do
                if b.type == "text" and type(b.text) == "string" then table.insert(parts, b.text) end
            end
        elseif config.provider == "Google" then
            local cand = parsed.candidates[1]
            finish = cand.finishReason
            for _, p in ipairs(cand.content and cand.content.parts or {}) do
                if type(p.text) == "string" and not p.thought then table.insert(parts, p.text) end
            end
        else
            local choice = parsed.choices[1]
            finish = choice.finish_reason
            local msg = choice.message or {}
            reasoning = msg.reasoning_content or msg.reasoning
            if type(msg.content) == "string" then
                table.insert(parts, msg.content)
            elseif type(msg.content) == "table" then
                for _, p in ipairs(msg.content) do
                    if type(p) == "table" and type(p.text) == "string" then table.insert(parts, p.text) end
                end
            end
        end
        text = table.concat(parts, "\n")
    end)

    if text then
        text = text:gsub("<think>.-</think>", ""):gsub("<think>.*$", "")
        text = text:gsub("^%s+", ""):gsub("%s+$", "")
    end

    local apiError = parsed.error and (type(parsed.error) == "table" and parsed.error.message or tostring(parsed.error)) or parsed.detail

    if (not text or text == "") and not apiError and maxTokens < 16384 then
        WindUI:Notify({ Title = "XAiK", Content = "Respuesta vacía, reintentando con más tokens...", Icon = "refresh-cw", Duration = 4 })
        return sendRequest(req, provider, model, retried, 16384)
    end

    if not text or text == "" then
        table.remove(history)
        if apiError then
            return nil, "[API " .. status .. "] " .. tostring(apiError)
        end
        local hint = ""
        if finish == "length" or finish == "max_tokens" or finish == "MAX_TOKENS" then
            hint = " El modelo agotó los tokens razonando; prueba otro modelo o un mensaje más corto."
        elseif finish == "SAFETY" or finish == "content_filter" then
            hint = " La respuesta fue bloqueada por el filtro del proveedor."
        elseif reasoning then
            hint = " El modelo solo devolvió razonamiento interno."
        end
        return nil, "[API " .. status .. "] Respuesta vacía (fin: " .. tostring(finish) .. ")." .. hint
    end

    table.insert(history, { role = "assistant", content = text })
    return text
end

local function escapeRich(s)
    return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local codeLangs = { [""] = true, lua = true, luau = true, rbxlua = true, roblox = true }

local function isCodeBlock(lang)
    return codeLangs[(lang or ""):lower()] == true
end

local filePattern = "%[%[%s*archivo%s*:%s*(.-)%s*%]%]%s*```[%w_]*[^\n]*\n(.-)```"

local function parseTools(text)
    local tools = {}
    text = text:gsub(filePattern, function(name, code)
        table.insert(tools, { kind = "archivo", arg = name, code = code })
        return ""
    end)
    for q in text:gmatch("%[%[%s*buscar%s*:%s*(.-)%s*%]%]") do table.insert(tools, { kind = "buscar", arg = q }) end
    for p in text:gmatch("%[%[%s*inspeccionar%s*:%s*(.-)%s*%]%]") do table.insert(tools, { kind = "inspeccionar", arg = p }) end
    if text:find("%[%[%s*remotes%s*%]%]") then table.insert(tools, { kind = "remotes" }) end
    for f in text:gmatch("%[%[%s*consola%s*:?%s*(.-)%s*%]%]") do table.insert(tools, { kind = "consola", arg = f }) end
    if text:find("%[%[%s*personaje%s*%]%]") then table.insert(tools, { kind = "personaje" }) end
    for f in text:gmatch("%[%[%s*scripts%s*:?%s*(.-)%s*%]%]") do table.insert(tools, { kind = "scripts", arg = f }) end
    for q in text:gmatch("%[%[%s*buscar_codigo%s*:%s*(.-)%s*%]%]") do table.insert(tools, { kind = "buscar_codigo", arg = q }) end
    for p in text:gmatch("%[%[%s*codigo%s*:%s*(.-)%s*%]%]") do table.insert(tools, { kind = "codigo", arg = p }) end
    for m in text:gmatch("%[%[%s*chat%s*:%s*(.-)%s*%]%]") do table.insert(tools, { kind = "chat", arg = m }) end
    for f in text:gmatch("%[%[%s*guardar_scripts%s*:?%s*(.-)%s*%]%]") do table.insert(tools, { kind = "guardar_scripts", arg = f }) end
    for lang, code in text:gmatch("```([%w_]*)[^\n]*\n(.-)```") do
        if isCodeBlock(lang) and code:find("%S") then
            table.insert(tools, { kind = "ejecutar", arg = code })
        end
    end
    return tools
end

local function cleanText(text)
    local t = text:gsub(filePattern, "")
    t = t:gsub("%[%[.-%]%]", "")
    t = t:gsub("```([%w_]*)([^\n]*\n.-)```", function(lang, rest)
        if isCodeBlock(lang) then return "" end
        return "```" .. lang .. rest .. "```"
    end)
    return (t:gsub("^%s+", ""):gsub("%s+$", ""))
end

local function destroyElement(e)
    if not e then return end
    if pcall(function() e:Destroy() end) then return end

        local ok, inst = pcall(function() return e[key] end)
        if ok and typeof(inst) == "Instance" then
            pcall(function() inst:Destroy() end)
            return
        end
    end
    pcall(function() e.UIElements.Main:Destroy() end)
end

local function renderCode(entry)
    local name, location, className, src = entry.name, entry.location, entry.className, entry.src
    local lineCount = select(2, src:gsub("\n", "\n")) + 1
    local chunks = {}
    local pos = 1
    while pos <= #src do
        local stop = math.min(pos + 3500, #src)
        if stop < #src then
            local nl = src:sub(pos, stop):match(".*()\n")
            if nl and nl > 200 then stop = pos + nl - 1 end
        end
        table.insert(chunks, src:sub(pos, stop))
        pos = stop + 1
        if #chunks >= 6 and pos <= #src then
            chunks[#chunks] = chunks[#chunks] .. "\n\n...(vista recortada para no saturar el dispositivo; usa Copiar script o Guardar archivo para tener el código completo)"
            break
        end
    end
    if #chunks == 0 then chunks = { "" } end
    local savedPath
    local buttons = {
        copyButton(src),
        {
            Title = "Guardar archivo",
            Icon = "save",
            Callback = function()
                savedPath = saveExport(name, src)
                WindUI:Notify({
                    Title = "XAiK",
                    Content = savedPath and ("Guardado en la carpeta workspace del executor: " .. savedPath) or "Tu executor no permite guardar archivos.",
                    Icon = savedPath and "save" or "triangle-alert",
                    Duration = 6,
                })
            end,
        },
    }
    for i, chunk in ipairs(chunks) do
        local header = ""
        if i == 1 then
            header = "Nombre: " .. name .. "\nRuta: " .. location .. "\nTipo: " .. className .. "\nLíneas: " .. lineCount .. "\n\n"
        end
        chatParagraph({
            Title = "Código: " .. name .. (#chunks > 1 and (" (parte " .. i .. "/" .. #chunks .. ")") or ""),
            Desc = escapeRich(header .. chunk),
            Image = "file-code",
            ImageSize = 20,
            Buttons = i == 1 and buttons or nil,
        })
    end
    return lineCount
end

local function renderEntry(entry)
    if entry.role == "code" then
        renderCode(entry)
        return nil
    end
    local isUser = entry.role == "user"
    local buttons
    if entry.pending then
        buttons = { executeButton(entry.pending), copyButton(entry.pending) }
    elseif entry.code then
        buttons = { copyButton(entry.code), executeButton(entry.code) }
    end
    return chatParagraph({
        Title = isUser and (LocalPlayer and LocalPlayer.DisplayName or "Tú") or (entry.role == "tool" and "XAiK · herramienta" or "XAiK"),
        Desc = entry.text,
        Image = isUser and "user" or (entry.role == "tool" and "wrench" or "brain"),
        ImageSize = 20,
        Color = isUser and "Blue" or nil,
        Buttons = buttons,
    })
end

end

local function loadChatsFromDisk()
    pcall(function()
        if not (listfiles and isfolder and isfolder(chatsFolder)) then return end
        for _, path in ipairs(listfiles(chatsFolder)) do
            if path:sub(-5) == ".json" then
                local ok, data = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
                if ok and type(data) == "table" and data.id and type(data.messages) == "table" then
                    chatCache[data.id] = data
                end
            end
        end
    end)
end

local function agentLoop(userText)
    local thinking = addMessage("ai", "Pensando...")
    local content = userText
    if config.allowScan then
        content = "[Contexto del jugador]\n" .. playerContext() .. "\n\n[Mensaje del usuario]\n" .. userText
    end

    for step = 1, 8 do
        local reply, err = callAPI(content)
        if not reply then
            pcall(function() thinking:SetDesc("Error: " .. tostring(err)) end)
            WindUI:Notify({ Title = "XAiK", Content = tostring(err), Icon = "triangle-alert", Duration = 5 })
            return
        end

        local tools = parseTools(reply)
        local visible = cleanText(reply)
        local skippedNote
        do
            local heavy = { buscar_codigo = true, guardar_scripts = true, remotes = true, scripts = true }
            local usedHeavy, kept, skipped = {}, {}, 0
            for _, tool in ipairs(tools) do
                if #kept >= perf().maxTools or (heavy[tool.kind] and usedHeavy[tool.kind]) then
                    skipped = skipped + 1
                else
                    if heavy[tool.kind] then usedHeavy[tool.kind] = true end
                    table.insert(kept, tool)
                end
            end
            if skipped > 0 then
                skippedNote = "[sistema] Se omitieron " .. skipped .. " herramientas para no sobrecargar el dispositivo. Máximo " .. perf().maxTools .. " por respuesta y solo una de cada tipo pesado (buscar_codigo, guardar_scripts, remotes, scripts). Pide las restantes en el siguiente paso si las necesitas."
            end
            tools = kept
        end

        if #tools == 0 then
            pcall(function() thinking:SetDesc(visible ~= "" and visible or reply) end)
            return
        end

        if visible ~= "" then
            pcall(function() thinking:SetDesc(visible) end)
        else
            pcall(function() thinking:SetDesc("Trabajando...") end)
        end

        local results = {}
        local pendingCode = {}
        for _, tool in ipairs(tools) do
            if tool.kind == "buscar" then
                if config.allowScan then
                    addMessage("tool", "Buscando: " .. tool.arg)
                    table.insert(results, "[buscar: " .. tool.arg .. "]\n" .. searchGame(tool.arg):sub(1, 5000))
                else
                    table.insert(results, "[buscar] El escaneo está desactivado por el usuario.")
                end
            elseif tool.kind == "inspeccionar" then
                if config.allowScan then
                    addMessage("tool", "Inspeccionando: " .. tool.arg)
                    table.insert(results, "[inspeccionar: " .. tool.arg .. "]\n" .. inspect(tool.arg):sub(1, 5000))
                else
                    table.insert(results, "[inspeccionar] El escaneo está desactivado por el usuario.")
                end
            elseif tool.kind == "remotes" then
                addMessage("tool", "Listando RemoteEvents y RemoteFunctions")
                table.insert(results, "[remotes]\n" .. listRemotes():sub(1, 5000))
            elseif tool.kind == "archivo" then
                local path = saveExport(tool.arg, tool.code)
                displayCode(tool.arg, path and ("workspace del executor/" .. path) or "Generado por XAiK", "Archivo", tool.code)
                table.insert(results, "[archivo: " .. tool.arg .. "] Mostrado al usuario" .. (path and (" y guardado en " .. path) or " (no se pudo guardar)") .. ".")
            elseif tool.kind == "chat" then
                local ok, info = Control.chat(tool.arg)
                addMessage("tool", (ok and "Mensaje enviado al chat: " or "No se pudo enviar al chat: ") .. tool.arg)
                table.insert(results, "[chat] " .. (ok and "Enviado: " .. tool.arg or "Falló: " .. tostring(info)))
            elseif not config.allowScan then
                table.insert(results, "[" .. tool.kind .. "] El escaneo está desactivado por el usuario.")
            elseif tool.kind == "consola" then
                addMessage("tool", "Revisando consola" .. (tool.arg ~= "" and (": " .. tool.arg) or ""))
                table.insert(results, "[consola" .. (tool.arg ~= "" and (": " .. tool.arg) or "") .. "]\n" .. getConsole(tool.arg):sub(1, 8000))
            elseif tool.kind == "personaje" then
                addMessage("tool", "Revisando tu personaje")
                table.insert(results, "[personaje]\n" .. characterReport():sub(1, 9000))
            elseif tool.kind == "scripts" then
                addMessage("tool", "Listando scripts" .. (tool.arg ~= "" and (": " .. tool.arg) or ""))
                table.insert(results, "[scripts" .. (tool.arg ~= "" and (": " .. tool.arg) or "") .. "]\n" .. listScripts(tool.arg):sub(1, 8000))
            elseif tool.kind == "buscar_codigo" then
                addMessage("tool", "Buscando en el código: " .. tool.arg)
                table.insert(results, "[buscar_codigo: " .. tool.arg .. "]\n" .. searchCode(tool.arg):sub(1, 6000))
            elseif tool.kind == "guardar_scripts" then
                addMessage("tool", "Guardando scripts del juego" .. (tool.arg ~= "" and (": " .. tool.arg) or "") .. "...")
                local report = dumpScripts(tool.arg)
                addMessage("tool", report:sub(1, 1500))
                table.insert(results, "[guardar_scripts]\n" .. report:sub(1, 4000))
            elseif tool.kind == "codigo" then
                local inst = resolvePath(tool.arg)
                local isScript = inst and select(2, pcall(function() return inst:IsA("LuaSourceContainer") end)) == true
                if not inst then
                    table.insert(results, "[codigo: " .. tool.arg .. "] No existe esa ruta. Usa [[scripts: nombre]] para encontrar la ruta exacta.")
                elseif not isScript then
                    table.insert(results, "[codigo: " .. tool.arg .. "] No es un script, es " .. inst.ClassName .. ".")
                else
                    local src = getSource(inst)
                    if not src then
                        addMessage("tool", "No se pudo leer el código de " .. inst:GetFullName())
                        table.insert(results, "[codigo: " .. tool.arg .. "] No se pudo obtener el código (" .. (decompile and "decompile falló" or "el executor no tiene decompile") .. ").")
                    else
                        local lines = displayCode(inst.Name, inst:GetFullName(), inst.ClassName, src)
                        table.insert(results, "[codigo: " .. inst:GetFullName() .. "] Mostrado completo al usuario (" .. lines .. " líneas). Código:\n" .. src:sub(1, 8000) .. (#src > 8000 and "\n...(recortado para ti, el usuario sí lo ve completo)" or ""))
                    end
                end
            elseif tool.kind == "ejecutar" then
                if config.autoExec then
                    local code = tool.arg
                    local ok, out = runCode(code)
                    addMessage("tool", (ok and "Script ejecutado\n" or "Fallo al ejecutar\n") .. out:sub(1, 500), { code = code })
                    table.insert(results, "[ejecutar]\n" .. out:sub(1, 3000))
                else
                    table.insert(pendingCode, tool.arg)
                end
            end
        end

        for _, code in ipairs(pendingCode) do
            addMessage("tool", "Script listo (ejecución automática desactivada)", { pending = code })
        end

        if skippedNote then table.insert(results, skippedNote) end
        if #results == 0 then return end

        thinking = addMessage("ai", "Analizando resultados...")
        content = "[Resultados de herramientas]\n" .. table.concat(results, "\n\n")
    end
    pcall(function() thinking:SetDesc("Se alcanzó el límite de pasos. Pide que continúe si hace falta.") end)
end

local function sendMessage()
    if processing then
        WindUI:Notify({ Title = "XAiK", Content = "Espera la respuesta anterior.", Icon = "loader", Duration = 3 })
        return
    end
    local text = currentMessage
    if text == "" then return end
    currentMessage = ""
    pcall(function() MessageInput:Set("") end)
    if currentChat.title == "Nueva conversación" then
        currentChat.title = text:gsub("%s+", " "):sub(1, 50)
    end
    addMessage("user", text)
    saveChat()
    processing = true
    task.spawn(function()
        local ok, err = pcall(agentLoop, text)
        if not ok then addMessage("ai", "Error interno: " .. tostring(err)) end
        processing = false
        saveChat()
        pcall(refreshHistory)
    end)
end

local function newChat()
    if processing then
        WindUI:Notify({ Title = "XAiK", Content = "Espera a que termine la respuesta.", Icon = "loader", Duration = 3 })
        return
    end
    saveChat()
    clearChatUI()
    startChat()
    renderEntry({ role = "ai", text = welcomeText })
    pcall(refreshHistory)
    WindUI:Notify({ Title = "XAiK", Content = "Nueva conversación iniciada.", Icon = "check", Duration = 3 })
end

local function openChat(id)
    if processing then
        WindUI:Notify({ Title = "XAiK", Content = "Espera a que termine la respuesta.", Icon = "loader", Duration = 3 })
        return
    end
    local chat = chatCache[id]
    if not chat then return end
    saveChat()
    clearChatUI()
    chat.history = chat.history or {}
    currentChat = chat
    history = chat.history
    for _, entry in ipairs(chat.messages) do
        pcall(renderEntry, entry)
    end
    pcall(refreshHistory)
    pcall(function() Window:SelectTab(1) end)
    WindUI:Notify({ Title = "XAiK", Content = "Conversación abierta: " .. tostring(chat.title), Icon = "folder-open", Duration = 3 })
end

local function deleteChat(id)
    if processing and currentChat and currentChat.id == id then return end
    chatCache[id] = nil
    pcall(function() delfile(chatsFolder .. "/" .. id .. ".json") end)
    if currentChat and currentChat.id == id then
        currentChat.messages = {}
        newChat()
    else
        pcall(refreshHistory)
    end
    WindUI:Notify({ Title = "XAiK", Content = "Conversación borrada.", Icon = "trash-2", Duration = 3 })
end

local historyElements = {}

function refreshHistory()
    for _, e in ipairs(historyElements) do destroyElement(e) end
    historyElements = {}
    local list = {}
    for _, c in pairs(chatCache) do table.insert(list, c) end
    table.sort(list, function(a, b) return (a.updated or 0) > (b.updated or 0) end)
    if #list == 0 then
        table.insert(historyElements, HistoryTab:Paragraph({
            Title = "Sin conversaciones",
            Desc = "Tus chats guardados aparecerán aquí.",
            Image = "inbox",
            ImageSize = 18,
        }))
        return
    end
    for i, c in ipairs(list) do
        if i > 60 then break end
        local id = c.id
        local userCount = 0
        for _, m in ipairs(c.messages or {}) do
            if m.role == "user" then userCount = userCount + 1 end
        end
        local isCurrent = currentChat and currentChat.id == id
        table.insert(historyElements, HistoryTab:Paragraph({
            Title = (isCurrent and "(Actual) " or "") .. tostring(c.title or "Conversación"),
            Desc = os.date("%d/%m/%Y %H:%M", c.updated or c.created or 0) .. "  |  " .. userCount .. " mensajes  |  " .. tostring(c.provider or ""),
            Image = "message-square",
            ImageSize = 18,
            Buttons = {
                { Title = "Abrir", Icon = "folder-open", Callback = function() openChat(id) end },
                { Title = "Borrar", Icon = "trash-2", Callback = function() deleteChat(id) end },
            },
        }))
    end
end

local function guiOnScreen(gui)
    local cur = gui
    while cur and cur:IsA("GuiObject") do
        if not cur.Visible then return false end
        if cur:IsA("CanvasGroup") and cur.GroupTransparency > 0.9 then return false end
        cur = cur.Parent
    end
    if cur and cur:IsA("LayerCollector") and not cur.Enabled then return false end
    local parent = gui.Parent
    if parent and parent:IsA("GuiObject") then
        if math.abs(gui.AbsolutePosition.X - parent.AbsolutePosition.X) > parent.AbsoluteSize.X * 0.5 then return false end
    end
    return gui.AbsoluteSize.X > 80 and gui.AbsoluteSize.Y > 80
end

            local p, s = scroll.AbsolutePosition, scroll.AbsoluteSize
            bar.Position = UDim2.fromOffset(p.X + 6, p.Y + s.Y - 56)
            bar.Size = UDim2.fromOffset(math.max(s.X - 12, 140), 50)
        elseif box:IsFocused() then
            box:ReleaseFocus()
        end
    end)
    return true
end
